using System; using System.Collections.Generic; // ========================================== // 1. КЛАСС СТУДЕНТА (Student.cs) // ========================================== public class Student { public string FirstName { get; set; } public string LastName { get; set; } public int Age { get; set; } public string Group { get; set; } public string Email { get; set; } public string GetFullName() { return FirstName + " " + LastName; } } // ========================================== // 2. КЛАСС КУРСА (Course.cs) // ========================================== public class Course { public string Title { get; set; } public string TeacherName { get; set; } } // ========================================== // 3. СЕРВИС СТУДЕНТОВ (StudentService.cs) // ========================================== public class StudentService { private List students = new List(); public bool AddStudent(Student student) { foreach (Student s in students) { if (s.GetFullName().ToLower() == student.GetFullName().ToLower()) { return false; } } students.Add(student); return true; } public List GetAllStudents() => students; public bool RemoveStudentByFullName(string fullName) { Student studentToRemove = null; foreach (Student s in students) { if (s.GetFullName().ToLower() == fullName.ToLower()) { studentToRemove = s; break; } } if (studentToRemove != null) { students.Remove(studentToRemove); return true; } return false; } public List GetStudentsByGroup(string groupName) { List result = new List(); foreach (Student s in students) { if (s.Group.ToLower() == groupName.ToLower()) result.Add(s); } return result; } public int GetStudentsCount() => students.Count; public double GetAverageAge() { if (students.Count == 0) return 0; double totalAge = 0; foreach (Student s in students) totalAge += s.Age; return totalAge / students.Count; } } // ========================================== // 4. СЕРВИС КУРСОВ (CourseService.cs) // ========================================== public class CourseService { private List courses = new List(); public void AddCourse(Course course) => courses.Add(course); public List GetAllCourses() => courses; } // ========================================== // 5. ГЛАВНЫЙ КЛАСС (Program.cs) // ========================================== class Program { static void Main() { StudentService studentService = new StudentService(); CourseService courseService = new CourseService(); studentService.AddStudent(new Student { FirstName = "Иван", LastName = "Киселев", Age = 20, Group = "ИКТп-53", Email = "ivan@pguti.ru" }); studentService.AddStudent(new Student { FirstName = "Мария", LastName = "Сидорова", Age = 19, Group = "ИКТп-54", Email = "maria@example.com" }); courseService.AddCourse(new Course { Title = "Языки программирования", TeacherName = "Иванов И.И." }); bool isRunning = true; while (isRunning) { Console.WriteLine("\n=================================="); Console.WriteLine("1 - Показать всех студентов (и их количество)"); Console.WriteLine("2 - Добавить студента"); Console.WriteLine("3 - Удалить студента по имени"); Console.WriteLine("4 - Показать студентов группы"); Console.WriteLine("5 - Показать средний возраст"); Console.WriteLine("6 - Показать все курсы"); Console.WriteLine("0 - Выход"); Console.Write("Выберите действие: "); string choice = Console.ReadLine(); switch (choice) { case "1": Console.WriteLine($"\n--- Всего студентов: {studentService.GetStudentsCount()} ---"); foreach (Student s in studentService.GetAllStudents()) Console.WriteLine($"{s.GetFullName()} | Группа: {s.Group} | Возраст: {s.Age} | Email: {s.Email}"); break; case "2": Console.Write("Введите имя: "); string firstName = Console.ReadLine(); Console.Write("Введите фамилию: "); string lastName = Console.ReadLine(); Console.Write("Введите возраст: "); int age = int.Parse(Console.ReadLine()); Console.Write("Введите группу: "); string group = Console.ReadLine(); Console.Write("Введите email: "); string email = Console.ReadLine(); Student newStudent = new Student { FirstName = firstName, LastName = lastName, Age = age, Group = group, Email = email }; if (studentService.AddStudent(newStudent)) Console.WriteLine("Студент успешно добавлен!"); else Console.WriteLine("Ошибка: Студент с таким именем уже есть!"); break; case "3": Console.Write("Введите имя и фамилию: "); string fullName = Console.ReadLine(); Console.WriteLine(studentService.RemoveStudentByFullName(fullName) ? "Удалено." : "Не найден."); break; case "4": Console.Write("Введите группу: "); string searchGroup = Console.ReadLine(); List groupStudents = studentService.GetStudentsByGroup(searchGroup); Console.WriteLine($"\n--- Студенты группы {searchGroup} ---"); if (groupStudents.Count == 0) Console.WriteLine("Нет студентов."); else foreach (Student s in groupStudents) Console.WriteLine($"{s.GetFullName()} (возраст: {s.Age})"); break; case "5": Console.WriteLine($"\nСредний возраст: {studentService.GetAverageAge():F1} лет"); break; case "6": Console.WriteLine("\n--- Список курсов ---"); foreach (Course c in courseService.GetAllCourses()) Console.WriteLine($"Курс: {c.Title} | Преподаватель: {c.TeacherName}"); break; case "0": isRunning = false; break; default: Console.WriteLine("Неизвестная команда."); break; } } } }
